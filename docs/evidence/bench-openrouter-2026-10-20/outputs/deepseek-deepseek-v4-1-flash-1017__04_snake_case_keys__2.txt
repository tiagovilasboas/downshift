def to_snake_case_keys(obj):
    import re

    def convert_key(key):
        if not isinstance(key, str):
            return key
        key = key.replace('-', '_')
        key = re.sub(r'(?<=[a-z0-9])([A-Z])', r'_\1', key)
        return key.lower()

    if isinstance(obj, dict):
        return {
            convert_key(key): to_snake_case_keys(value)
            for key, value in obj.items()
        }
    if isinstance(obj, list):
        return [to_snake_case_keys(item) for item in obj]
    return obj
