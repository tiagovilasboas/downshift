def merge_intervals(intervals):
    if not intervals:
        return []

    sorted_intervals = sorted((list(interval) for interval in intervals), key=lambda x: x[0])
    merged = []

    for start, end in sorted_intervals:
        if not merged or start > merged[-1][1]:
            merged.append([start, end])
        else:
            if end > merged[-1][1]:
                merged[-1][1] = end

    return merged
