def call_with_retry(fn, max_attempts, base_delay, sleep):
    if max_attempts < 1:
        raise ValueError("max_attempts must be at least 1")

    for attempt in range(max_attempts):
        try:
            return fn()
        except Exception:
            if attempt == max_attempts - 1:
                raise
            sleep(base_delay * (2 ** attempt))
