def json_get(data, path, default=None):
    if path == "":
        return data

    current = data

    for segment in path.split('.'):
        bracket = segment.find('[')

        if bracket == -1:
            key = segment
            indexes = []
        else:
            key = segment[:bracket]
            rest = segment[bracket:]
            indexes = []
            i = 0

            while i < len(rest):
                if rest[i] != '[':
                    return default

                end = rest.find(']', i + 1)
                if end == -1:
                    return default

                index_str = rest[i + 1:end]
                if not index_str.isdigit():
                    return default

                indexes.append(int(index_str))
                i = end + 1

        if key != "":
            if not isinstance(current, dict) or key not in current:
                return default
            current = current[key]

        for index in indexes:
            if not isinstance(current, list) or index < 0 or index >= len(current):
                return default
            current = current[index]

    return current
