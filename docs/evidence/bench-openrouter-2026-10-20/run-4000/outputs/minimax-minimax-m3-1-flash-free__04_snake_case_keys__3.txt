def to_snake_case_keys(obj):
    def convert_key(key):
        if not isinstance(key, str):
            return key

        result = []
        previous = None

        for char in key:
            if char == "-":
                result.append("_")
            else:
                if char.isupper() and (
                    previous is not None
                    and (previous.islower() or previous.isdigit())
                ):
                    result.append("_")
                result.append(char.lower())

            previous = char

        return "".join(result)

    if isinstance(obj, dict):
        return {
            convert_key(key): to_snake_case_keys(value)
            for key, value in obj.items()
        }

    if isinstance(obj, list):
        return [to_snake_case_keys(value) for value in obj]

    return obj
