package task

import (
	"fmt"
	"sort"
)

func TopoSort(deps map[string][]string) ([]string, error) {
	// Build in-degree map and adjacency list (reverse dependency)
	inDegree := make(map[string]int)
	reverseDeps := make(map[string][]string) // dependsOn -> list of dependents

	// Initialize nodes with zero in-degree
	for node := range deps {
		if _, exists := inDegree[node]; !exists {
			inDegree[node] = 0
		}
	}

	// Build reverse dependency graph
	for node, prerequisites := range deps {
		for _, prereq := range prerequisites {
			// Ensure prereq exists in inDegree
			if _, exists := inDegree[prereq]; !exists {
				inDegree[prereq] = 0
			}
			reverseDeps[prereq] = append(reverseDeps[prereq], node)
			inDegree[node]++
		}
	}

	// Collect all unique nodes
	allNodes := make(map[string]bool)
	for node := range deps {
		allNodes[node] = true
		for _, prereq := range deps[node] {
			allNodes[prereq] = true
		}
	}

	// Initialize min-heap (priority queue) with zero in-degree nodes
	queue := make([]string, 0)
	for node := range allNodes {
		if inDegree[node] == 0 {
			queue = append(queue, node)
		}
	}
	// Sort alphabetically for deterministic selection
	sort.Strings(queue)

	result := make([]string, 0, len(allNodes))

	for len(queue) > 0 {
		// Pop the alphabetically smallest node
		current := queue[0]
		queue = queue[1:]
		result = append(result, current)

		// Decrease in-degree for each dependent
		for _, neighbor := range reverseDeps[current] {
			inDegree[neighbor]--
			if inDegree[neighbor] == 0 {
				// Insert in sorted order
				insertIdx := sort.SearchStrings(queue, neighbor)
				queue = append(queue, "")
				copy(queue[insertIdx+1:], queue[insertIdx:])
				queue[insertIdx] = neighbor
			}
		}
	}

	if len(result) != len(allNodes) {
		return nil, fmt.Errorf("cycle detected in dependencies")
	}

	return result, nil
}
