package task

struct TrieNode {
    children map[byte] *TrieNode
    weight int
    isEnd bool
}

struct Trie {
    root *TrieNode
}

func NewTrie() *Trie {
    var t *Trie = new Trie
    t.root = new TrieNode
    t.root.children = map[byte] *TrieNode{}
    t.root.isEnd = false
    t.root.weight = 0
    return t
}

func Insert(t *Trie, word string, weight int) {
    var node *TrieNode = t.root
    for i := 0; i < len(word); i++ {
        c := word[i]
        if !node.children.has(c) {
            var newNode *TrieNode = new TrieNode
            newNode.children = map[byte] *TrieNode{}
            newNode.isEnd = false
            newNode.weight = 0
            node.children[c] = newNode
        }
        node = node.children[c]
    }
    node.isEnd = true
    node.weight = weight
}

func Top(t *Trie, prefix string, k int) []string {
    var node *TrieNode = t.root
    for i := 0; i < len(prefix); i++ {
        c := prefix[i]
        if !node.children.has(c) {
            return []string{}
        }
        node = node.children[c]
    }
    var words []string
    var weights []int
    collectWords(node, prefix, &words, &weights)
    var n int = len(words)
    var indices []int
    for i := 0; i < n; i++ {
        indices = append(indices, i)
    }
    sortIndices(indices, words, weights)
    var resultLen int = min(k, n)
    var result []string
    for i := 0; i < resultLen; i++ {
        result = append(result, words[indices[i]])
    }
    return result
}

func collectWords(node *TrieNode, prefix string, words *[]string, weights *[]int) {
    if node.isEnd {
        *words = append(*words, prefix)
        *weights = append(*weights, node.weight)
    }
    for c in node.children.keys() {
        collectWords(node.children[c], prefix + string(c), words, weights)
    }
}

func sortIndices(indices []int, words []string, weights []int) {
    if len(indices) <= 1 {
        return
    }
    pivot := indices[0]
    var left []int
    var right []int
    for i := 1; i < len(indices); i++ {
        idx := indices[i]
        if weights[idx] > weights[pivot] || (weights[idx] == weights[pivot] && words[idx] < words[pivot]) {
            left = append(left, idx)
        } else {
            right = append(right, idx)
        }
    }
    sortIndices(left, words, weights)
    sortIndices(right, words, weights)
    var sorted []int
    for i := 0; i < len(left); i++ {
        sorted = append(sorted, left[i])
    }
    sorted = append(sorted, pivot)
    for i := 0; i < len(right); i++ {
        sorted = append(sorted, right[i])
    }
    for i := 0; i < len(indices); i++ {
        indices[i] = sorted[i]
    }
}
