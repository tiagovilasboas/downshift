package task

import (
	"context"
	"sync"
)

func ParallelMap(ctx context.Context, inputs []int, workers int, fn func(context.Context, int) (int, error)) ([]int, error) {
	if workers <= 0 {
		workers = 1
	}
	if len(inputs) == 0 {
		return []int{}, nil
	}

	results := make([]int, len(inputs))
	errChan := make(chan error, 1)
	ctx, cancel := context.WithCancel(ctx)
	defer cancel()

	var wg sync.WaitGroup
	inputCh := make(chan struct {
		idx int
		val int
	})

	// Start worker goroutines
	for i := 0; i < workers; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for job := range inputCh {
				select {
				case <-ctx.Done():
					return
				default:
				}
				result, err := fn(ctx, job.val)
				if err != nil {
					select {
					case errChan <- err:
					default:
					}
					cancel()
					return
				}
				results[job.idx] = result
			}
		}()
	}

	// Send jobs
	for i, val := range inputs {
		select {
		case <-ctx.Done():
			break
		default:
		}
		select {
		case inputCh <- struct {
			idx int
			val int
		}{i, val}:
		case <-ctx.Done():
			break
		}
	}

	close(inputCh)
	wg.Wait()

	select {
	case err := <-errChan:
		return nil, err
	default:
		return results, nil
	}
}
