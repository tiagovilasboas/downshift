package task

import (
	"fmt"
	"strconv"
	"strings"
	"time"
)

type field struct {
	values map[int]bool
}

func parseField(s string, min, max int) (*field, error) {
	f := &field{values: make(map[int]bool)}
	if s == "*" {
		for i := min; i <= max; i++ {
			f.values[i] = true
		}
		return f, nil
	}
	parts := strings.Split(s, ",")
	for _, p := range parts {
		if strings.Contains(p, "/") {
			stepParts := strings.Split(p, "/")
			if len(stepParts) != 2 {
				return nil, fmt.Errorf("invalid step: %s", p)
			}
			step, err := strconv.Atoi(stepParts[1])
			if err != nil || step <= 0 {
				return nil, fmt.Errorf("invalid step value: %s", stepParts[1])
			}
			rangePart := stepParts[0]
			var start, end int
			if rangePart == "*" {
				start = min
				end = max
			} else if strings.Contains(rangePart, "-") {
				rangeParts := strings.Split(rangePart, "-")
				if len(rangeParts) != 2 {
					return nil, fmt.Errorf("invalid range: %s", rangePart)
				}
				start, err = strconv.Atoi(rangeParts[0])
				if err != nil {
					return nil, fmt.Errorf("invalid number: %s", rangeParts[0])
				}
				end, err = strconv.Atoi(rangeParts[1])
				if err != nil {
					return nil, fmt.Errorf("invalid number: %s", rangeParts[1])
				}
			} else {
				start, err = strconv.Atoi(rangePart)
				if err != nil {
					return nil, fmt.Errorf("invalid number: %s", rangePart)
				}
				end = start
			}
			if start < min || end > max || start > end {
				return nil, fmt.Errorf("range out of bounds: %s", rangePart)
			}
			for i := start; i <= end; i += step {
				f.values[i] = true
			}
		} else if strings.Contains(p, "-") {
			rangeParts := strings.Split(p, "-")
			if len(rangeParts) != 2 {
				return nil, fmt.Errorf("invalid range: %s", p)
			}
			start, err := strconv.Atoi(rangeParts[0])
			if err != nil {
				return nil, fmt.Errorf("invalid number: %s", rangeParts[0])
			}
			end, err := strconv.Atoi(rangeParts[1])
			if err != nil {
				return nil, fmt.Errorf("invalid number: %s", rangeParts[1])
			}
			if start < min || end > max || start > end {
				return nil, fmt.Errorf("range out of bounds: %s", p)
			}
			for i := start; i <= end; i++ {
				f.values[i] = true
			}
		} else {
			v, err := strconv.Atoi(p)
			if err != nil {
				return nil, fmt.Errorf("invalid number: %s", p)
			}
			if v < min || v > max {
				return nil, fmt.Errorf("value out of bounds: %s", p)
			}
			f.values[v] = true
		}
	}
	return f, nil
}

func Next(spec string, after time.Time) (time.Time, error) {
	fields := strings.Fields(spec)
	if len(fields) != 5 {
		return time.Time{}, fmt.Errorf("invalid cron spec: expected 5 fields, got %d", len(fields))
	}

	minuteField, err := parseField(fields[0], 0, 59)
	if err != nil {
		return time.Time{}, fmt.Errorf("minute: %v", err)
	}
	hourField, err := parseField(fields[1], 0, 23)
	if err != nil {
		return time.Time{}, fmt.Errorf("hour: %v", err)
	}
	dayField, err := parseField(fields[2], 1, 31)
	if err != nil {
		return time.Time{}, fmt.Errorf("day-of-month: %v", err)
	}
	monthField, err := parseField(fields[3], 1, 12)
	if err != nil {
		return time.Time{}, fmt.Errorf("month: %v", err)
	}
	weekdayField, err := parseField(fields[4], 0, 6)
	if err != nil {
		return time.Time{}, fmt.Errorf("day-of-week: %v", err)
	}

	loc := after.Location()
	t := after.Truncate(time.Minute).Add(time.Minute)

	// Check if day-of-month and day-of-week are both restricted (not all values)
	domRestricted := len(dayField.values) < 31
	dowRestricted := len(weekdayField.values) < 7

	for i := 0; i < 366*24*60; i++ { // search up to 1 year
		y, m, d := t.Date()
		wd := int(t.Weekday())
		domOk := dayField.values[d]
		dowOk := weekdayField.values[wd]
		dayOk := true
		if domRestricted && dowRestricted {
			dayOk = domOk || dowOk
		} else if domRestricted {
			dayOk = domOk
		} else if dowRestricted {
			dayOk = dowOk
		}

		if monthField.values[int(m)] && dayOk && hourField.values[t.Hour()] && minuteField.values[t.Minute()] {
			return t, nil
		}
		t = t.Add(time.Minute)
	}
	return time.Time{}, fmt.Errorf("no matching time found within one year")
}
