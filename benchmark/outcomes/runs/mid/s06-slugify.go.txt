package task

import "strings"

func Slugify(s string) string {
	s = strings.ToLower(s)
	var b strings.Builder
	dash := false

	for _, r := range s {
		if (r >= 'a' && r <= 'z') || (r >= '0' && r <= '9') {
			if dash && b.Len() > 0 {
				b.WriteByte('-')
			}
			b.WriteRune(r)
			dash = false
		} else {
			dash = true
		}
	}

	return b.String()
}
