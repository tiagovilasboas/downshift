package task

import (
	"errors"
	"strconv"
)

func EvalRPN(tokens []string) (int, error) {
	if len(tokens) == 0 {
		return 0, errors.New("empty expression")
	}

	stack := make([]int, 0, len(tokens))

	for _, token := range tokens {
		if num, err := strconv.Atoi(token); err == nil {
			stack = append(stack, num)
			continue
		}

		if len(stack) < 2 {
			return 0, errors.New("malformed expression: insufficient operands")
		}

		b := stack[len(stack)-1]
		a := stack[len(stack)-2]
		stack = stack[:len(stack)-2]

		var result int
		switch token {
		case "+":
			result = a + b
		case "-":
			result = a - b
		case "*":
			result = a * b
		case "/":
			if b == 0 {
				return 0, errors.New("division by zero")
			}
			result = a / b
		default:
			return 0, errors.New("unknown operator: " + token)
		}
		stack = append(stack, result)
	}

	if len(stack) != 1 {
		return 0, errors.New("malformed expression: too many operands")
	}

	return stack[0], nil
}
