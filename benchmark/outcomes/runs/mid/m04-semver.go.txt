package task

import (
	"errors"
	"strconv"
	"strings"
)

// CompareSemver compares two semantic version strings.
// Returns -1 if a < b, 0 if equal, 1 if a > b.
// Returns an error if either version is invalid.
func CompareSemver(a, b string) (int, error) {
	ma, mi, pa, preA, err := parseVersion(a)
	if err != nil {
		return 0, err
	}
	mb, mi2, pb, preB, err := parseVersion(b)
	if err != nil {
		return 0, err
	}

	// Compare major, minor, patch
	if ma != mb {
		if ma < mb {
			return -1, nil
		}
		return 1, nil
	}
	if mi != mi2 {
		if mi < mi2 {
			return -1, nil
		}
		return 1, nil
	}
	if pa != pb {
		if pa < pb {
			return -1, nil
		}
		return 1, nil
	}

	// Compare prerelease
	return comparePrerelease(preA, preB), nil
}

// parseVersion splits a semver string into its components.
func parseVersion(v string) (major, minor, patch int, prerelease []string, err error) {
	// Split into core and optional prerelease
	var core string
	if idx := strings.IndexByte(v, '-'); idx >= 0 {
		core = v[:idx]
		prereleaseStr := v[idx+1:]
		if prereleaseStr == "" {
			err = errors.New("invalid semver: empty prerelease")
			return
		}
		prerelease = strings.Split(prereleaseStr, ".")
		// Validate each prerelease identifier
		for _, id := range prerelease {
			if id == "" {
				err = errors.New("invalid semver: empty prerelease identifier")
				return
			}
			if !isValidPrereleaseID(id) {
				err = errors.New("invalid semver: invalid prerelease identifier")
				return
			}
		}
	} else {
		core = v
	}

	// Split core into major.minor.patch
	parts := strings.Split(core, ".")
	if len(parts) != 3 {
		err = errors.New("invalid semver: must have exactly three numeric parts")
		return
	}

	major, err = parseNumeric(parts[0])
	if err != nil {
		return
	}
	minor, err = parseNumeric(parts[1])
	if err != nil {
		return
	}
	patch, err = parseNumeric(parts[2])
	if err != nil {
		return
	}
	return
}

// parseNumeric parses a string as a non-negative integer without leading zeros.
func parseNumeric(s string) (int, error) {
	if len(s) == 0 {
		return 0, errors.New("invalid semver: empty numeric part")
	}
	if len(s) > 1 && s[0] == '0' {
		return 0, errors.New("invalid semver: leading zeros not allowed")
	}
	n, err := strconv.Atoi(s)
	if err != nil || n < 0 {
		return 0, errors.New("invalid semver: non-numeric part")
	}
	return n, nil
}

// isValidPrereleaseID checks if a prerelease identifier is valid.
// It can be numeric (no leading zeros) or alphanumeric (contains only [0-9A-Za-z-]).
func isValidPrereleaseID(id string) bool {
	if len(id) == 0 {
		return false
	}
	// Check if all digits
	allDigits := true
	for _, ch := range id {
		if ch < '0' || ch > '9' {
			allDigits = false
			break
		}
	}
	if allDigits {
		// Must not have leading zeros
		if len(id) > 1 && id[0] == '0' {
			return false
		}
		return true
	}
	// Otherwise must be alphanumeric with hyphens
	for _, ch := range id {
		if !((ch >= '0' && ch <= '9') || (ch >= 'A' && ch <= 'Z') || (ch >= 'a' && ch <= 'z') || ch == '-') {
			return false
		}
	}
	return true
}

// comparePrerelease compares two prerelease identifier lists.
// Returns -1 if a < b, 0 if equal, 1 if a > b.
func comparePrerelease(a, b []string) int {
	// No prerelease vs prerelease: no prerelease is greater
	if len(a) == 0 && len(b) == 0 {
		return 0
	}
	if len(a) == 0 {
		return 1 // a has no prerelease, so a > b
	}
	if len(b) == 0 {
		return -1 // b has no prerelease, so a < b
	}

	// Compare identifier by identifier
	minLen := len(a)
	if len(b) < minLen {
		minLen = len(b)
	}
	for i := 0; i < minLen; i++ {
		cmp := compareIdentifiers(a[i], b[i])
		if cmp != 0 {
			return cmp
		}
	}
	// All compared identifiers equal; shorter list is less
	if len(a) < len(b) {
		return -1
	}
	if len(a) > len(b) {
		return 1
	}
	return 0
}

// compareIdentifiers compares two prerelease identifiers.
// Numeric identifiers have lower precedence than alphanumeric.
// Numeric comparison is numerical; alphanumeric is lexicographic (ASCII).
func compareIdentifiers(a, b string) int {
	aNum := isNumeric(a)
	bNum := isNumeric(b)

	if aNum && bNum {
		// Both numeric: compare numerically
		na, _ := strconv.Atoi(a)
		nb, _ := strconv.Atoi(b)
		if na < nb {
			return -1
		}
		if na > nb {
			return 1
		}
		return 0
	}
	if aNum && !bNum {
		// Numeric < alphanumeric
		return -1
	}
	if !aNum && bNum {
		// Alphanumeric > numeric
		return 1
	}
	// Both alphanumeric: lexicographic comparison
	if a < b {
		return -1
	}
	if a > b {
		return 1
	}
	return 0
}

// isNumeric checks if a string consists only of digits.
func isNumeric(s string) bool {
	for _, ch := range s {
		if ch < '0' || ch > '9' {
			return false
		}
	}
	return true
}
