package task

// Match reports whether the whole string s matches the pattern.
// The pattern supports:
//   - literal characters
//   - '.' (matches any single character)
//   - postfix operators '*', '+', '?' applied to the preceding character or '.'
func Match(pattern, s string) bool {
	return matchHelper(pattern, 0, s, 0)
}

// matchHelper recursively checks if pattern[pi:] matches s[si:].
func matchHelper(p string, pi int, s string, si int) bool {
	// If pattern is exhausted, we must have consumed the whole string.
	if pi == len(p) {
		return si == len(s)
	}

	// Check if the next character is a postfix operator.
	hasOp := pi+1 < len(p) && (p[pi+1] == '*' || p[pi+1] == '+' || p[pi+1] == '?')

	if !hasOp {
		// No operator: match a single character.
		if si >= len(s) {
			return false
		}
		if p[pi] == '.' || p[pi] == s[si] {
			return matchHelper(p, pi+1, s, si+1)
		}
		return false
	}

	// Has operator.
	c := p[pi]
	op := p[pi+1]

	switch op {
	case '*':
		// Zero or more: try zero first, then one or more recursively.
		if matchHelper(p, pi+2, s, si) {
			return true
		}
		if si < len(s) && (c == '.' || c == s[si]) {
			if matchHelper(p, pi, s, si+1) {
				return true
			}
		}
		return false

	case '+':
		// One or more: must match at least one.
		if si >= len(s) || !(c == '.' || c == s[si]) {
			return false
		}
		// After matching one, either stop or continue.
		if matchHelper(p, pi+2, s, si+1) {
			return true
		}
		if matchHelper(p, pi, s, si+1) {
			return true
		}
		return false

	case '?':
		// Zero or one: try zero first, then one.
		if matchHelper(p, pi+2, s, si) {
			return true
		}
		if si < len(s) && (c == '.' || c == s[si]) {
			if matchHelper(p, pi+2, s, si+1) {
				return true
			}
		}
		return false
	}

	return false
}
