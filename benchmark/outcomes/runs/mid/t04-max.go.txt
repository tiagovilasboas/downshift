package task

struct MaxIntResult {
    max int
    ok bool
}

func MaxInt(xs []int) MaxIntResult {
    if len(xs) == 0 {
        return MaxIntResult{max: 0, ok: false}
    }
    maxVal := xs[0]
    for i := 1; i < len(xs); i++ {
        if xs[i] > maxVal {
            maxVal = xs[i]
        }
    }
    return MaxIntResult{max: maxVal, ok: true}
}
