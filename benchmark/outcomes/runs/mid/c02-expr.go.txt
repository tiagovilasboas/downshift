package task

import (
	"fmt"
	"strconv"
	"strings"
	"unicode"
)

type lexer struct {
	input []byte
	pos   int
}

type token struct {
	typ tokenType
	val string
}

type tokenType int

const (
	tokNumber tokenType = iota
	tokPlus
	tokMinus
	tokMul
	tokDiv
	tokLParen
	tokRParen
	tokEOF
	tokError
)

func (l *lexer) next() token {
	for l.pos < len(l.input) && unicode.IsSpace(rune(l.input[l.pos])) {
		l.pos++
	}
	if l.pos >= len(l.input) {
		return token{typ: tokEOF}
	}

	c := l.input[l.pos]

	if c == '+' {
		l.pos++
		return token{typ: tokPlus, val: "+"}
	}
	if c == '-' {
		l.pos++
		return token{typ: tokMinus, val: "-"}
	}
	if c == '*' {
		l.pos++
		return token{typ: tokMul, val: "*"}
	}
	if c == '/' {
		l.pos++
		return token{typ: tokDiv, val: "/"}
	}
	if c == '(' {
		l.pos++
		return token{typ: tokLParen, val: "("}
	}
	if c == ')' {
		l.pos++
		return token{typ: tokRParen, val: ")"}
	}

	if c == '.' || unicode.IsDigit(rune(c)) {
		start := l.pos
		for l.pos < len(l.input) && (l.input[l.pos] == '.' || unicode.IsDigit(rune(l.input[l.pos]))) {
			l.pos++
		}
		return token{typ: tokNumber, val: string(l.input[start:l.pos])}
	}

	return token{typ: tokError, val: string(c)}
}

type parser struct {
	lex   *lexer
	curr  token
	peek  token
	err   error
}

func newParser(expr string) *parser {
	p := &parser{lex: &lexer{input: []byte(expr)}}
	p.advance()
	p.advance()
	return p
}

func (p *parser) advance() {
	p.curr = p.peek
	p.peek = p.lex.next()
}

func (p *parser) parseExpr() (float64, error) {
	if p.err != nil {
		return 0, p.err
	}
	result := p.parseTerm()
	if p.err != nil {
		return 0, p.err
	}
	for p.peek.typ == tokPlus || p.peek.typ == tokMinus {
		op := p.peek.typ
		p.advance()
		right := p.parseTerm()
		if p.err != nil {
			return 0, p.err
		}
		switch op {
		case tokPlus:
			result += right
		case tokMinus:
			result -= right
		}
	}
	if p.curr.typ == tokError {
		return 0, fmt.Errorf("unexpected character: %q", p.curr.val)
	}
	return result, nil
}

func (p *parser) parseTerm() float64 {
	result := p.parseFactor()
	if p.err != nil {
		return 0
	}
	for p.peek.typ == tokMul || p.peek.typ == tokDiv {
		op := p.peek.typ
		p.advance()
		right := p.parseFactor()
		if p.err != nil {
			return 0
		}
		if op == tokDiv {
			if right == 0 {
				p.err = fmt.Errorf("division by zero")
				return 0
			}
			result /= right
		} else {
			result *= right
		}
	}
	return result
}

func (p *parser) parseFactor() float64 {
	if p.peek.typ == tokMinus {
		p.advance()
		val := p.parseFactor()
		if p.err != nil {
			return 0
		}
		return -val
	}
	if p.peek.typ == tokLParen {
		p.advance() // consume '('
		p.advance() // move past '('
		result := p.parseExpr()
		if p.err != nil {
			return 0
		}
		if p.curr.typ != tokRParen {
			p.err = fmt.Errorf("missing closing parenthesis")
			return 0
		}
		p.advance() // consume ')'
		return result
	}
	if p.peek.typ == tokNumber {
		p.advance()
		val, err := strconv.ParseFloat(p.curr.val, 64)
		if err != nil {
			p.err = fmt.Errorf("invalid number: %s", p.curr.val)
			return 0
		}
		return val
	}
	p.err = fmt.Errorf("unexpected token: %s", p.peek.val)
	return 0
}

func Eval(expr string) (float64, error) {
	p := newParser(expr)
	result := p.parseExpr()
	if p.err != nil {
		// Check for division by zero error
		if strings.Contains(p.err.Error(), "division by zero") {
			return 0, p.err
		}
		// Check for unexpected characters at end
		if p.curr.typ == tokError {
			return 0, fmt.Errorf("unexpected character: %q", p.curr.val)
		}
		return 0, p.err
	}
	// Ensure we consumed all input
	if p.curr.typ != tokEOF {
		return 0, fmt.Errorf("unexpected input at end")
	}
	return result, nil
}
