package task

func Spiral(m [][]int) []int {
	if len(m) == 0 || len(m[0]) == 0 {
		return []int{}
	}

	top, bottom := 0, len(m)-1
	left, right := 0, len(m[0])-1
	result := make([]int, 0, len(m)*len(m[0]))

	for top <= bottom && left <= right {
		// Traverse right along top row
		for i := left; i <= right; i++ {
			result = append(result, m[top][i])
		}
		top++

		// Traverse down along right column
		for i := top; i <= bottom; i++ {
			result = append(result, m[i][right])
		}
		right--

		// Traverse left along bottom row (if still valid)
		if top <= bottom {
			for i := right; i >= left; i-- {
				result = append(result, m[bottom][i])
			}
			bottom--
		}

		// Traverse up along left column (if still valid)
		if left <= right {
			for i := bottom; i >= top; i-- {
				result = append(result, m[i][left])
			}
			left++
		}
	}

	return result
}
