package task

import (
    "strings"
    "unicode"
)

func WordFreq(text string) map[string]int {
    words := strings.FieldsFunc(text, func(r rune) bool {
        return !unicode.IsLetter(r)
    })

    freq := make(map[string]int)
    for _, w := range words {
        w = strings.ToLower(w)
        freq[w]++
    }
    return freq
}
