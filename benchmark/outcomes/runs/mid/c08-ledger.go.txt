package task

import (
	"errors"
	"sync"
)

// Ledger is a thread-safe ledger that tracks account balances.
type Ledger struct {
	mu       sync.RWMutex
	balances map[string]int64
}

// NewLedger creates a new Ledger initialized with the given balances.
// The input map is copied to avoid external mutations.
func NewLedger(balances map[string]int64) *Ledger {
	copy := make(map[string]int64, len(balances))
	for k, v := range balances {
		copy[k] = v
	}
	return &Ledger{balances: copy}
}

// Balance returns the current balance of the given account.
// It returns 0 if the account does not exist (though the spec says unknown accounts are rejected in Transfer).
func (l *Ledger) Balance(account string) int64 {
	l.mu.RLock()
	defer l.mu.RUnlock()
	return l.balances[account]
}

// Transfer moves amount from account 'from' to account 'to'.
// It returns an error if:
//   - amount is not positive
//   - either account does not exist
//   - from and to are the same
//   - from has insufficient funds
func (l *Ledger) Transfer(from, to string, amount int64) error {
	if amount <= 0 {
		return errors.New("amount must be positive")
	}
	if from == to {
		return errors.New("cannot transfer to self")
	}

	l.mu.Lock()
	defer l.mu.Unlock()

	// Check existence of both accounts
	fromBal, fromExists := l.balances[from]
	_, toExists := l.balances[to]
	if !fromExists || !toExists {
		return errors.New("unknown account")
	}

	// Check overdraft
	if fromBal < amount {
		return errors.New("insufficient funds")
	}

	// Perform transfer
	l.balances[from] -= amount
	l.balances[to] += amount
	return nil
}
