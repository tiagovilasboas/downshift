package task

// Diff computes a minimal line diff between a and b based on the longest common subsequence.
// Each output line is prefixed with "  " (unchanged), "- " (only in a), or "+ " (only in b).
// The unchanged plus removed lines reconstruct a, and the unchanged plus added lines reconstruct b.
func Diff(a, b []string) []string {
	// Compute LCS length table
	m, n := len(a), len(b)
	dp := make([][]int, m+1)
	for i := 0; i <= m; i++ {
		dp[i] = make([]int, n+1)
	}
	for i := 1; i <= m; i++ {
		for j := 1; j <= n; j++ {
			if a[i-1] == b[j-1] {
				dp[i][j] = dp[i-1][j-1] + 1
			} else {
				if dp[i-1][j] > dp[i][j-1] {
					dp[i][j] = dp[i-1][j]
				} else {
					dp[i][j] = dp[i][j-1]
				}
			}
		}
	}

	// Backtrack to build diff lines (in reverse order)
	var result []string
	i, j := m, n
	for i > 0 || j > 0 {
		if i > 0 && j > 0 && a[i-1] == b[j-1] {
			result = append(result, "  "+a[i-1])
			i--
			j--
		} else if i > 0 && (j == 0 || dp[i-1][j] >= dp[i][j-1]) {
			result = append(result, "- "+a[i-1])
			i--
		} else {
			result = append(result, "+ "+b[j-1])
			j--
		}
	}

	// Reverse the result to get correct order
	for left, right := 0, len(result)-1; left < right; left, right = left+1, right-1 {
		result[left], result[right] = result[right], result[left]
	}
	return result
}
