package task

import (
	"errors"
	"sort"
)

func TopoSort(deps map[string][]string) ([]string, error) {
	// Calculate in-degrees for each node
	inDegree := make(map[string]int)
	allNodes := make(map[string]bool)
	
	// Initialize all nodes
	for node := range deps {
		allNodes[node] = true
		for _, dep := range deps[node] {
			allNodes[dep] = true
		}
	}
	
	// Calculate in-degrees
	for _, dependencies := range deps {
		for _, dep := range dependencies {
			inDegree[dep]++
		}
	}
	
	// Find all nodes with in-degree 0
	queue := []string{}
	for node := range allNodes {
		if inDegree[node] == 0 {
			queue = append(queue, node)
		}
	}
	
	// Sort queue alphabetically
	sort.Strings(queue)
	
	result := []string{}
	
	for len(queue) > 0 {
		// Take the first element (alphabetically smallest)
		current := queue[0]
		queue = queue[1:]
		result = append(result, current)
		
		// Remove current node and update in-degrees
		for node, dependencies := range deps {
			for i, dep := range dependencies {
				if dep == current {
					// Remove dependency
					deps[node] = append(dependencies[:i], dependencies[i+1:]...)
					inDegree[node]--
					
					// If in-degree becomes 0, add to queue
					if inDegree[node] == 0 {
						queue = append(queue, node)
					}
					break
				}
			}
		}
		
		// Sort queue alphabetically again
		sort.Strings(queue)
	}
	
	// Check for cycles - if we haven't processed all nodes, there's a cycle
	if len(result) != len(allNodes) {
		return nil, errors.New("cycle detected")
	}
	
	return result, nil
}
