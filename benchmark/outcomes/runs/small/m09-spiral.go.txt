package task

func Spiral(m [][]int) []int {
	if len(m) == 0 || len(m[0]) == 0 {
		return []int{}
	}

	result := make([]int, 0)
	
	top, bottom := 0, len(m)-1
	left, right := 0, len(m[0])-1
	
	for top <= bottom && left <= right {
		// Traverse right along top row
		for col := left; col <= right; col++ {
			result = append(result, m[top][col])
		}
		top++
		
		// Traverse down along right column
		for row := top; row <= bottom; row++ {
			result = append(result, m[row][right])
		}
		right--
		
		// Traverse left along bottom row (if there's still a row to traverse)
		if top <= bottom {
			for col := right; col >= left; col-- {
				result = append(result, m[bottom][col])
			}
			bottom--
		}
		
		// Traverse up along left column (if there's still a column to traverse)
		if left <= right {
			for row := bottom; row >= top; row-- {
				result = append(result, m[row][left])
			}
			left++
		}
	}
	
	return result
}
