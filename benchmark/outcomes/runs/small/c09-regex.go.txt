package task

func Match(pattern, s string) bool {
	if pattern == "" {
		return s == ""
	}
	
	// Check if the pattern matches the entire string
	return matchHelper(pattern, s, 0, 0)
}

func matchHelper(pattern, s string, pIndex, sIndex int) bool {
	// If we've reached the end of the pattern
	if pIndex == len(pattern) {
		// We should also be at the end of the string
		return sIndex == len(s)
	}
	
	// Check if next character is '*'
	if pIndex+1 < len(pattern) && pattern[pIndex+1] == '*' {
		// Two cases: either match zero occurrences or match one and continue
		// Match zero occurrences
		if matchHelper(pattern, s, pIndex+2, sIndex) {
			return true
		}
		// Match one occurrence and continue with same pattern
		if (sIndex < len(s)) && (pattern[pIndex] == '.' || pattern[pIndex] == s[sIndex]) {
			return matchHelper(pattern, s, pIndex, sIndex+1)
		}
		return false
	}
	
	// Check if next character is '+'
	if pIndex+1 < len(pattern) && pattern[pIndex+1] == '+' {
		// Must match at least one occurrence
		if (sIndex < len(s)) && (pattern[pIndex] == '.' || pattern[pIndex] == s[sIndex]) {
			// Match one occurrence and continue with same pattern
			if matchHelper(pattern, s, pIndex, sIndex+1) {
				return true
			}
			// Or match zero occurrences (but this is impossible due to +)
			// Actually, we need to match at least one, so we continue with same pattern
			// But we already handled the case of matching one, so we need to check if we can match more
			// Let's restructure this logic
			return matchHelper(pattern, s, pIndex+2, sIndex+1)
		}
		return false
	}
	
	// Check if next character is '?'
	if pIndex+1 < len(pattern) && pattern[pIndex+1] == '?' {
		// Match zero or one occurrence
		// Zero occurrence case
		if matchHelper(pattern, s, pIndex+2, sIndex) {
			return true
		}
		// One occurrence case
		if (sIndex < len(s)) && (pattern[pIndex] == '.' || pattern[pIndex] == s[sIndex]) {
			return matchHelper(pattern, s, pIndex+2, sIndex+1)
		}
		return false
	}
	
	// Regular character matching
	if (sIndex < len(s)) && (pattern[pIndex] == '.' || pattern[pIndex] == s[sIndex]) {
		return matchHelper(pattern, s, pIndex+1, sIndex+1)
	}
	
	return false
}
