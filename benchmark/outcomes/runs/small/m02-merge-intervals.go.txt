package task

import (
	"sort"
)

func Merge(intervals [][2]int) [][2]int {
	if len(intervals) == 0 {
		return nil
	}
	
	// Create a copy to avoid mutating the input
	copied := make([][2]int, len(intervals))
	copy(copied, intervals)
	
	// Sort by start time
	sort.Slice(copied, func(i, j int) bool {
		return copied[i][0] < copied[j][0]
	})
	
	// Merge overlapping intervals
	var result [][2]int
	currentStart, currentEnd := copied[0][0], copied[0][1]
	
	for i := 1; i < len(copied); i++ {
		start, end := copied[i][0], copied[i][1]
		
		if start <= currentEnd {
			// Overlapping or touching, merge them
			if end > currentEnd {
				currentEnd = end
			}
		} else {
			// Non-overlapping, add the previous interval to result
			result = append(result, [2]int{currentStart, currentEnd})
			currentStart, currentEnd = start, end
		}
	}
	
	// Add the last interval
	result = append(result, [2]int{currentStart, currentEnd})
	
	return result
}
