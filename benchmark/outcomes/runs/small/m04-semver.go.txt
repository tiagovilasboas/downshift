package task

import (
	"errors"
	"strconv"
	"strings"
)

func CompareSemver(a, b string) (int, error) {
	partsA, err := parseVersion(a)
	if err != nil {
		return 0, err
	}
	
	partsB, err := parseVersion(b)
	if err != nil {
		return 0, err
	}
	
	// Compare major versions
	if partsA.major > partsB.major {
		return 1, nil
	}
	if partsA.major < partsB.major {
		return -1, nil
	}
	
	// Compare minor versions
	if partsA.minor > partsB.minor {
		return 1, nil
	}
	if partsA.minor < partsB.minor {
		return -1, nil
	}
	
	// Compare patch versions
	if partsA.patch > partsB.patch {
		return 1, nil
	}
	if partsA.patch < partsB.patch {
		return -1, nil
	}
	
	// If we get here, the numeric parts are equal
	// Now compare prerelease identifiers
	return comparePrerelease(partsA.prerelease, partsB.prerelease), nil
}

type version struct {
	major     int
	minor     int
	patch     int
	prerelease []string
}

func parseVersion(v string) (*version, error) {
	// Split on - to separate prerelease
	parts := strings.Split(v, "-")
	
	// Parse the main version part (MAJOR.MINOR.PATCH)
	mainParts := strings.Split(parts[0], ".")
	if len(mainParts) != 3 {
		return nil, errors.New("invalid version format")
	}
	
	major, err := strconv.Atoi(mainParts[0])
	if err != nil {
		return nil, errors.New("invalid major version")
	}
	
	minor, err := strconv.Atoi(mainParts[1])
	if err != nil {
		return nil, errors.New("invalid minor version")
	}
	
	patch, err := strconv.Atoi(mainParts[2])
	if err != nil {
		return nil, errors.New("invalid patch version")
	}
	
	prerelease := []string{}
	if len(parts) > 1 {
		prerelease = strings.Split(parts[1], ".")
	}
	
	return &version{
		major:     major,
		minor:     minor,
		patch:     patch,
		prerelease: prerelease,
	}, nil
}

func comparePrerelease(a, b []string) int {
	// If a is empty and b is not, a comes first (prerelease sorts before release)
	if len(a) == 0 && len(b) > 0 {
		return -1
	}
	// If b is empty and a is not, b comes first
	if len(b) == 0 && len(a) > 0 {
		return 1
	}
	// If both are empty, they're equal
	if len(a) == 0 && len(b) == 0 {
		return 0
	}
	
	// Compare each identifier in order
	for i := 0; i < len(a) && i < len(b); i++ {
		// Try to parse as integers
		numA, errA := strconv.Atoi(a[i])
		numB, errB := strconv.Atoi(b[i])
		
		// If both are numeric, compare numerically
		if errA == nil && errB == nil {
			if numA > numB {
				return 1
			}
			if numA < numB {
				return -1
			}
			continue
		}
		
		// If only a is numeric, it comes first
		if errA == nil {
			return -1
		}
		
		// If only b is numeric, it comes first
		if errB == nil {
			return 1
		}
		
		// Both are strings, compare lexically
		if a[i] > b[i] {
			return 1
		}
		if a[i] < b[i] {
			return -1
		}
	}
	
	// If we get here, the common parts are equal
	// The shorter one comes first
	if len(a) > len(b) {
		return 1
	}
	if len(a) < len(b) {
		return -1
	}
	
	return 0
}
