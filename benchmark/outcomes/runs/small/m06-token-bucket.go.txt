package task

import (
	"time"
)

type Bucket struct {
	capacity     int64
	refillPerSec float64
	now          func() time.Time
	tokens       int64
	lastRefill   time.Time
}

func NewBucket(capacity int, refillPerSec float64, now func() time.Time) *Bucket {
	return &Bucket{
		capacity:     int64(capacity),
		refillPerSec: refillPerSec,
		now:          now,
		tokens:       int64(capacity),
		lastRefill:   now(),
	}
}

func (b *Bucket) Allow() bool {
	now := b.now()
	
	// Calculate time elapsed since last refill
	elapsed := now.Sub(b.lastRefill)
	
	// Calculate how many tokens to add
	if elapsed > 0 {
		tokensToAdd := int64(float64(elapsed) / float64(time.Second) * b.refillPerSec)
		if tokensToAdd > 0 {
			b.tokens = min(b.capacity, b.tokens+tokensToAdd)
			b.lastRefill = now
		}
	}
	
	// Check if we can allow this request
	if b.tokens >= 1 {
		b.tokens--
		return true
	}
	
	return false
}

func min(a, b int64) int64 {
	if a < b {
		return a
	}
	return b
}
