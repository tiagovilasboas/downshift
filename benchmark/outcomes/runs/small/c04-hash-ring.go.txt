package task

import (
	"crypto/md5"
	"fmt"
	"sort"
	"sync"
)

type Ring struct {
	replicas int
	points   []int
	nodes    map[int]string
	nodeMap  map[string]bool
	mutex    sync.RWMutex
}

func NewRing(replicas int) *Ring {
	return &Ring{
		replicas: replicas,
		points:   make([]int, 0),
		nodes:    make(map[int]string),
		nodeMap:  make(map[string]bool),
	}
}

func (r *Ring) Add(node string) {
	r.mutex.Lock()
	defer r.mutex.Unlock()

	if r.nodeMap[node] {
		return
	}

	r.nodeMap[node] = true
	for i := 0; i < r.replicas; i++ {
		point := r.hash(fmt.Sprintf("%s%d", node, i))
		r.points = append(r.points, point)
		r.nodes[point] = node
	}
	sort.Ints(r.points)
}

func (r *Ring) Remove(node string) {
	r.mutex.Lock()
	defer r.mutex.Unlock()

	if !r.nodeMap[node] {
		return
	}

	delete(r.nodeMap, node)
	newPoints := make([]int, 0)
	for _, point := range r.points {
		if r.nodes[point] != node {
			newPoints = append(newPoints, point)
		} else {
			delete(r.nodes, point)
		}
	}
	r.points = newPoints
	sort.Ints(r.points)
}

func (r *Ring) Get(key string) string {
	r.mutex.RLock()
	defer r.mutex.RUnlock()

	if len(r.points) == 0 {
		return ""
	}

	point := r.hash(key)
	index := sort.Search(len(r.points), func(i int) bool {
		return r.points[i] >= point
	})

	if index == len(r.points) {
		index = 0
	}

	return r.nodes[r.points[index]]
}

func (r *Ring) hash(key string) int {
	hash := md5.Sum([]byte(key))
	return int(hash[0])<<24 | int(hash[1])<<16 | int(hash[2])<<8 | int(hash[3])
}
