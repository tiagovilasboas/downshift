package task

func Diff(a, b []string) []string {
	if len(a) == 0 && len(b) == 0 {
		return []string{}
	}
	if len(a) == 0 {
		result := make([]string, len(b))
		for i, line := range b {
			result[i] = "+ " + line
		}
		return result
	}
	if len(b) == 0 {
		result := make([]string, len(a))
		for i, line := range a {
			result[i] = "- " + line
		}
		return result
	}

	// Build LCS table
	m, n := len(a), len(b)
	dp := make([][]int, m+1)
	for i := range dp {
		dp[i] = make([]int, n+1)
	}

	for i := 1; i <= m; i++ {
		for j := 1; j <= n; j++ {
			if a[i-1] == b[j-1] {
				dp[i][j] = dp[i-1][j-1] + 1
			} else {
				if dp[i-1][j] > dp[i][j-1] {
					dp[i][j] = dp[i-1][j]
				} else {
					dp[i][j] = dp[i][j-1]
				}
			}
		}
	}

	// Reconstruct LCS and build diff
	result := []string{}
	i, j := m, n

	for i > 0 || j > 0 {
		if i > 0 && j > 0 && a[i-1] == b[j-1] {
			// Line is common
			result = append([]string{"  " + a[i-1]}, result...)
			i--
			j--
		} else if i > 0 && (j == 0 || dp[i-1][j] > dp[i][j-1]) {
			// Line only in a
			result = append([]string{"- " + a[i-1]}, result...)
			i--
		} else {
			// Line only in b
			result = append([]string{"+ " + b[j-1]}, result...)
			j--
		}
	}

	return result
}
