package task

import (
	"errors"
	"sync"
)

var (
	ErrInvalidAmount     = errors.New("ledger: amount must be positive")
	ErrUnknownAccount    = errors.New("ledger: unknown account")
	ErrSameAccount       = errors.New("ledger: cannot transfer to the same account")
	ErrInsufficientFunds = errors.New("ledger: insufficient funds")
)

type account struct {
	mu      sync.Mutex
	balance int64
}

// Ledger is safe for concurrent use. The set of accounts is fixed at
// construction time, so the map itself is read-only afterwards; each
// account's balance is protected by its own mutex.
type Ledger struct {
	accounts map[string]*account
}

func NewLedger(balances map[string]int64) *Ledger {
	l := &Ledger{accounts: make(map[string]*account, len(balances))}
	for name, b := range balances {
		l.accounts[name] = &account{balance: b}
	}
	return l
}

func (l *Ledger) Balance(name string) int64 {
	a, ok := l.accounts[name]
	if !ok {
		return 0
	}
	a.mu.Lock()
	defer a.mu.Unlock()
	return a.balance
}

func (l *Ledger) Transfer(from, to string, amount int64) error {
	if amount <= 0 {
		return ErrInvalidAmount
	}
	src, ok := l.accounts[from]
	if !ok {
		return ErrUnknownAccount
	}
	dst, ok := l.accounts[to]
	if !ok {
		return ErrUnknownAccount
	}
	if from == to {
		return ErrSameAccount
	}

	// Lock in a consistent global order (by account name) to avoid deadlock.
	first, second := src, dst
	if to < from {
		first, second = dst, src
	}
	first.mu.Lock()
	defer first.mu.Unlock()
	second.mu.Lock()
	defer second.mu.Unlock()

	if src.balance < amount {
		return ErrInsufficientFunds
	}
	src.balance -= amount
	dst.balance += amount
	return nil
}
