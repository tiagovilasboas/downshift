package task

// Chunk splits xs into consecutive chunks of size n. The last chunk may be
// shorter than n. It returns nil when n <= 0.
func Chunk(xs []int, n int) [][]int {
	if n <= 0 {
		return nil
	}
	var out [][]int
	for i := 0; i < len(xs); i += n {
		end := i + n
		if end > len(xs) || end < i {
			end = len(xs)
		}
		chunk := make([]int, end-i)
		copy(chunk, xs[i:end])
		out = append(out, chunk)
	}
	return out
}
