package task

import (
	"errors"
	"fmt"
	"strings"
)

type semver struct {
	core [3]string
	pre  []string
}

func isDigits(s string) bool {
	if s == "" {
		return false
	}
	for i := 0; i < len(s); i++ {
		if s[i] < '0' || s[i] > '9' {
			return false
		}
	}
	return true
}

func validIdentChars(s string) bool {
	if s == "" {
		return false
	}
	for i := 0; i < len(s); i++ {
		c := s[i]
		if !((c >= '0' && c <= '9') || (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || c == '-') {
			return false
		}
	}
	return true
}

func validNumeric(s string) bool {
	return isDigits(s) && !(len(s) > 1 && s[0] == '0')
}

func parse(v string) (semver, error) {
	var sv semver
	if v == "" {
		return sv, errors.New("empty version")
	}
	// Strip build metadata (ignored for precedence).
	if i := strings.IndexByte(v, '+'); i >= 0 {
		build := v[i+1:]
		v = v[:i]
		if build == "" {
			return sv, fmt.Errorf("invalid build metadata")
		}
		for _, id := range strings.Split(build, ".") {
			if !validIdentChars(id) {
				return sv, fmt.Errorf("invalid build identifier %q", id)
			}
		}
	}
	core := v
	if i := strings.IndexByte(v, '-'); i >= 0 {
		core = v[:i]
		pre := v[i+1:]
		if pre == "" {
			return sv, errors.New("empty prerelease")
		}
		for _, id := range strings.Split(pre, ".") {
			if !validIdentChars(id) {
				return sv, fmt.Errorf("invalid prerelease identifier %q", id)
			}
			if isDigits(id) && !validNumeric(id) {
				return sv, fmt.Errorf("leading zero in prerelease identifier %q", id)
			}
			sv.pre = append(sv.pre, id)
		}
	}
	parts := strings.Split(core, ".")
	if len(parts) != 3 {
		return sv, fmt.Errorf("invalid version core %q", core)
	}
	for i, p := range parts {
		if !validNumeric(p) {
			return sv, fmt.Errorf("invalid numeric component %q", p)
		}
		sv.core[i] = p
	}
	return sv, nil
}

func cmpNum(a, b string) int {
	if len(a) != len(b) {
		if len(a) < len(b) {
			return -1
		}
		return 1
	}
	if a < b {
		return -1
	}
	if a > b {
		return 1
	}
	return 0
}

func cmpStr(a, b string) int {
	if a < b {
		return -1
	}
	if a > b {
		return 1
	}
	return 0
}

func CompareSemver(a, b string) (int, error) {
	va, err := parse(a)
	if err != nil {
		return 0, fmt.Errorf("invalid version %q: %w", a, err)
	}
	vb, err := parse(b)
	if err != nil {
		return 0, fmt.Errorf("invalid version %q: %w", b, err)
	}
	for i := 0; i < 3; i++ {
		if c := cmpNum(va.core[i], vb.core[i]); c != 0 {
			return c, nil
		}
	}
	if len(va.pre) == 0 && len(vb.pre) == 0 {
		return 0, nil
	}
	if len(va.pre) == 0 {
		return 1, nil
	}
	if len(vb.pre) == 0 {
		return -1, nil
	}
	for i := 0; i < len(va.pre) && i < len(vb.pre); i++ {
		x, y := va.pre[i], vb.pre[i]
		xn, yn := isDigits(x), isDigits(y)
		var c int
		switch {
		case xn && yn:
			c = cmpNum(x, y)
		case xn:
			c = -1
		case yn:
			c = 1
		default:
			c = cmpStr(x, y)
		}
		if c != 0 {
			return c, nil
		}
	}
	switch {
	case len(va.pre) < len(vb.pre):
		return -1, nil
	case len(va.pre) > len(vb.pre):
		return 1, nil
	}
	return 0, nil
}
