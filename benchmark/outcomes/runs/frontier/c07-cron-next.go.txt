package task

import (
	"errors"
	"fmt"
	"strconv"
	"strings"
	"time"
)

type field struct {
	bits [64]bool
	star bool
}

func parseNum(s string, lo, hi int) (int, error) {
	if s == "" {
		return 0, errors.New("empty number")
	}
	for _, c := range s {
		if c < '0' || c > '9' {
			return 0, fmt.Errorf("invalid number %q", s)
		}
	}
	n, err := strconv.Atoi(s)
	if err != nil {
		return 0, err
	}
	if n < lo || n > hi {
		return 0, fmt.Errorf("value %d out of range [%d,%d]", n, lo, hi)
	}
	return n, nil
}

func parseField(s string, lo, hi int) (*field, error) {
	f := &field{}
	if s == "" {
		return nil, errors.New("empty field")
	}
	f.star = s[0] == '*'
	for _, part := range strings.Split(s, ",") {
		if part == "" {
			return nil, errors.New("empty list element")
		}
		rng := part
		step := 1
		hasStep := false
		if i := strings.Index(part, "/"); i >= 0 {
			rng = part[:i]
			st, err := parseNum(part[i+1:], 1, 1<<30)
			if err != nil {
				return nil, err
			}
			step = st
			hasStep = true
		}
		var a, b int
		if rng == "*" {
			a, b = lo, hi
		} else if i := strings.Index(rng, "-"); i >= 0 {
			var err error
			a, err = parseNum(rng[:i], lo, hi)
			if err != nil {
				return nil, err
			}
			b, err = parseNum(rng[i+1:], lo, hi)
			if err != nil {
				return nil, err
			}
			if a > b {
				return nil, fmt.Errorf("invalid range %q", rng)
			}
		} else {
			var err error
			a, err = parseNum(rng, lo, hi)
			if err != nil {
				return nil, err
			}
			b = a
			if hasStep {
				b = hi
			}
		}
		for v := a; v <= b; v += step {
			f.bits[v] = true
		}
	}
	return f, nil
}

func Next(spec string, after time.Time) (time.Time, error) {
	parts := strings.Fields(spec)
	if len(parts) != 5 {
		return time.Time{}, errors.New("cron spec must have 5 fields")
	}
	minF, err := parseField(parts[0], 0, 59)
	if err != nil {
		return time.Time{}, err
	}
	hourF, err := parseField(parts[1], 0, 23)
	if err != nil {
		return time.Time{}, err
	}
	domF, err := parseField(parts[2], 1, 31)
	if err != nil {
		return time.Time{}, err
	}
	monF, err := parseField(parts[3], 1, 12)
	if err != nil {
		return time.Time{}, err
	}
	dowF, err := parseField(parts[4], 0, 7)
	if err != nil {
		return time.Time{}, err
	}
	if dowF.bits[7] {
		dowF.bits[0] = true
	}

	loc := after.Location()
	t := after.In(loc)
	t = t.Add(-time.Duration(t.Second())*time.Second - time.Duration(t.Nanosecond())).Add(time.Minute)
	limit := after.Year() + 10

	dayMatch := func(t time.Time) bool {
		d := domF.bits[t.Day()]
		w := dowF.bits[int(t.Weekday())]
		if domF.star || dowF.star {
			return d && w
		}
		return d || w
	}

	for t.Year() <= limit {
		if !monF.bits[int(t.Month())] {
			t = time.Date(t.Year(), t.Month()+1, 1, 0, 0, 0, 0, loc)
			continue
		}
		if !dayMatch(t) {
			t = time.Date(t.Year(), t.Month(), t.Day()+1, 0, 0, 0, 0, loc)
			continue
		}
		if !hourF.bits[t.Hour()] {
			t = t.Add(time.Duration(60-t.Minute()) * time.Minute)
			continue
		}
		if !minF.bits[t.Minute()] {
			t = t.Add(time.Minute)
			continue
		}
		return t, nil
	}
	return time.Time{}, errors.New("no matching time found")
}
