package task

import "strings"

// ParseKV parses semicolon-separated key=value pairs such as "a=1; b = 2".
// Keys and values are trimmed of surrounding whitespace. Entries without '='
// or with an empty key are skipped. Later duplicates overwrite earlier ones.
func ParseKV(s string) map[string]string {
	result := make(map[string]string)
	for _, part := range strings.Split(s, ";") {
		idx := strings.Index(part, "=")
		if idx < 0 {
			continue
		}
		key := strings.TrimSpace(part[:idx])
		if key == "" {
			continue
		}
		value := strings.TrimSpace(part[idx+1:])
		result[key] = value
	}
	return result
}
