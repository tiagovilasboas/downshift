package task

import "strconv"

// FizzBuzz returns the classic FizzBuzz sequence for 1..n.
// For multiples of 3 it uses "Fizz", for multiples of 5 "Buzz",
// for multiples of both "FizzBuzz", and otherwise the number itself.
// If n <= 0, an empty slice is returned.
func FizzBuzz(n int) []string {
	if n <= 0 {
		return []string{}
	}
	result := make([]string, 0, n)
	for i := 1; i <= n; i++ {
		switch {
		case i%15 == 0:
			result = append(result, "FizzBuzz")
		case i%3 == 0:
			result = append(result, "Fizz")
		case i%5 == 0:
			result = append(result, "Buzz")
		default:
			result = append(result, strconv.Itoa(i))
		}
	}
	return result
}
