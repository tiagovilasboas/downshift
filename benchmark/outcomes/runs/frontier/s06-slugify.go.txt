package task

import "strings"

// Slugify lowercases s, keeps ASCII letters and digits, collapses every run
// of other characters into a single dash, and trims leading and trailing dashes.
func Slugify(s string) string {
	var b strings.Builder
	b.Grow(len(s))
	pendingDash := false
	for i := 0; i < len(s); i++ {
		c := s[i]
		if c >= 'A' && c <= 'Z' {
			c += 'a' - 'A'
		}
		if (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') {
			if pendingDash && b.Len() > 0 {
				b.WriteByte('-')
			}
			pendingDash = false
			b.WriteByte(c)
		} else {
			pendingDash = true
		}
	}
	return b.String()
}
