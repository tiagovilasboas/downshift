package task

type token struct {
	c   rune
	any bool
	q   rune // 0, '*', '+', '?'
}

// Match reports whether the whole string s matches pattern.
// Supported: literal characters, '.', and postfix '*', '+', '?'.
// A quantifier with no preceding atom (or following another quantifier)
// is treated as a literal character.
func Match(pattern, s string) bool {
	var toks []token
	for _, r := range pattern {
		if (r == '*' || r == '+' || r == '?') && len(toks) > 0 && toks[len(toks)-1].q == 0 {
			toks[len(toks)-1].q = r
			continue
		}
		toks = append(toks, token{c: r, any: r == '.'})
	}

	str := []rune(s)
	n, m := len(toks), len(str)

	dp := make([][]bool, n+1)
	for i := range dp {
		dp[i] = make([]bool, m+2)
	}
	dp[n][m] = true

	for i := n - 1; i >= 0; i-- {
		t := toks[i]
		for j := m; j >= 0; j-- {
			ok := j < m && (t.any || str[j] == t.c)
			switch t.q {
			case '*':
				dp[i][j] = dp[i+1][j] || (ok && dp[i][j+1])
			case '+':
				dp[i][j] = ok && (dp[i+1][j+1] || dp[i][j+1])
			case '?':
				dp[i][j] = dp[i+1][j] || (ok && dp[i+1][j+1])
			default:
				dp[i][j] = ok && dp[i+1][j+1]
			}
		}
	}
	return dp[0][0]
}
