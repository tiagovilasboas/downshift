package task

import (
	"errors"
	"fmt"
	"strconv"
)

type parser struct {
	s   string
	pos int
}

// Eval evaluates an arithmetic expression supporting + - * /, parentheses,
// unary minus (and plus), decimal numbers and whitespace.
func Eval(expr string) (float64, error) {
	p := &parser{s: expr}
	p.skipSpace()
	if p.pos >= len(p.s) {
		return 0, errors.New("empty expression")
	}
	v, err := p.parseExpr()
	if err != nil {
		return 0, err
	}
	p.skipSpace()
	if p.pos < len(p.s) {
		return 0, fmt.Errorf("unexpected character %q at position %d", p.s[p.pos], p.pos)
	}
	return v, nil
}

func (p *parser) skipSpace() {
	for p.pos < len(p.s) {
		switch p.s[p.pos] {
		case ' ', '\t', '\n', '\r', '\f', '\v':
			p.pos++
		default:
			return
		}
	}
}

func (p *parser) peek() (byte, bool) {
	p.skipSpace()
	if p.pos >= len(p.s) {
		return 0, false
	}
	return p.s[p.pos], true
}

func (p *parser) parseExpr() (float64, error) {
	left, err := p.parseTerm()
	if err != nil {
		return 0, err
	}
	for {
		c, ok := p.peek()
		if !ok || (c != '+' && c != '-') {
			return left, nil
		}
		p.pos++
		right, err := p.parseTerm()
		if err != nil {
			return 0, err
		}
		if c == '+' {
			left += right
		} else {
			left -= right
		}
	}
}

func (p *parser) parseTerm() (float64, error) {
	left, err := p.parseUnary()
	if err != nil {
		return 0, err
	}
	for {
		c, ok := p.peek()
		if !ok || (c != '*' && c != '/') {
			return left, nil
		}
		opPos := p.pos
		p.pos++
		right, err := p.parseUnary()
		if err != nil {
			return 0, err
		}
		if c == '*' {
			left *= right
		} else {
			if right == 0 {
				return 0, fmt.Errorf("division by zero at position %d", opPos)
			}
			left /= right
		}
	}
}

func (p *parser) parseUnary() (float64, error) {
	c, ok := p.peek()
	if ok && (c == '-' || c == '+') {
		p.pos++
		v, err := p.parseUnary()
		if err != nil {
			return 0, err
		}
		if c == '-' {
			return -v, nil
		}
		return v, nil
	}
	return p.parsePrimary()
}

func (p *parser) parsePrimary() (float64, error) {
	c, ok := p.peek()
	if !ok {
		return 0, errors.New("unexpected end of expression")
	}
	if c == '(' {
		p.pos++
		v, err := p.parseExpr()
		if err != nil {
			return 0, err
		}
		c, ok = p.peek()
		if !ok || c != ')' {
			return 0, fmt.Errorf("expected ')' at position %d", p.pos)
		}
		p.pos++
		return v, nil
	}
	if (c >= '0' && c <= '9') || c == '.' {
		return p.parseNumber()
	}
	return 0, fmt.Errorf("unexpected character %q at position %d", c, p.pos)
}

func (p *parser) parseNumber() (float64, error) {
	start := p.pos
	digits := 0
	for p.pos < len(p.s) && p.s[p.pos] >= '0' && p.s[p.pos] <= '9' {
		p.pos++
		digits++
	}
	if p.pos < len(p.s) && p.s[p.pos] == '.' {
		p.pos++
		for p.pos < len(p.s) && p.s[p.pos] >= '0' && p.s[p.pos] <= '9' {
			p.pos++
			digits++
		}
	}
	if digits == 0 {
		return 0, fmt.Errorf("invalid number at position %d", start)
	}
	v, err := strconv.ParseFloat(p.s[start:p.pos], 64)
	if err != nil {
		return 0, fmt.Errorf("invalid number %q at position %d", p.s[start:p.pos], start)
	}
	return v, nil
}
