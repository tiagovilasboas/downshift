package task

import (
	"errors"
	"strings"
)

// ParseCSVLine splits a single CSV record into its fields.
// Fields may be enclosed in double quotes, in which case they may contain
// commas and escaped quotes (""). An unterminated quoted field results in
// an error, as does unexpected content after a closing quote.
func ParseCSVLine(line string) ([]string, error) {
	var fields []string
	var sb strings.Builder
	i := 0
	n := len(line)

	for {
		sb.Reset()
		if i < n && line[i] == '"' {
			// Quoted field.
			i++
			closed := false
			for i < n {
				c := line[i]
				if c == '"' {
					if i+1 < n && line[i+1] == '"' {
						sb.WriteByte('"')
						i += 2
						continue
					}
					i++
					closed = true
					break
				}
				sb.WriteByte(c)
				i++
			}
			if !closed {
				return nil, errors.New("unterminated quoted field")
			}
			fields = append(fields, sb.String())
			if i == n {
				return fields, nil
			}
			if line[i] != ',' {
				return nil, errors.New("unexpected character after closing quote")
			}
			i++ // skip comma
			continue
		}

		// Unquoted field.
		for i < n && line[i] != ',' {
			sb.WriteByte(line[i])
			i++
		}
		fields = append(fields, sb.String())
		if i == n {
			return fields, nil
		}
		i++ // skip comma
	}
}
